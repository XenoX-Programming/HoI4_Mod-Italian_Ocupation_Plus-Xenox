# ============================================================
# Italian Occupation Plus v0.0.16 - North Africa
# 1) As Italy, control Tripoli (starts Italian - INA can be
#    founded on day one).
# 2) Found INA with the "Establish" decision.
#    Founding requires the capital state to be controlled by Italy or one of its subjects (448 Tripoli).
#    Any other initial states controlled by Italy or its subjects transfer automatically.
# 3) Use the grouped "Transfer ..." decisions to hand further
#    regions straight to INA (cores + transfer, no event - INA
#    is the only recipient). Each needs INA to exist and Italy
#    to control at least one state of the group; every controlled
#    state of the group transfers.
# 4) Sidi Ifni (783) is special: its decision fires an event to
#    give it to INA OR integrate it into Italy proper.
# NOTE: none of these redistributed states has any other fate
# decision - each region is handled exactly once, here.
# ============================================================

iop_occupation = {

	# ------------------------------------------------------------
	# FOUNDING DECISION
	# ------------------------------------------------------------

	iop_establish_north_africa = {
		icon = generic_political_pressure

		highlight_states = {
			highlight_state_targets = {
				state = 448
				state = 661
				state = 449
				state = 662
				state = 450
				state = 451
				state = 663
				state = 273
			}
		}

		allowed = {
			original_tag = ITA
		}

		visible = {
			original_tag = ITA
			NOT = { country_exists = INA }
			# Only appears once Italy or its subjects control at least half of the required territory (4 of 8 states).
			count_triggers = {
				amount = 4
				448 = { iop_med_controlled = yes }
				661 = { iop_med_controlled = yes }
				449 = { iop_med_controlled = yes }
				662 = { iop_med_controlled = yes }
				450 = { iop_med_controlled = yes }
				451 = { iop_med_controlled = yes }
				663 = { iop_med_controlled = yes }
				273 = { iop_med_controlled = yes }
			}
		}

		available = {
			448 = { iop_med_controlled = yes }
		}

		fire_only_once = no

		complete_effect = {
			hidden_effect = {
				iop_grant_zone_north_africa = yes
			}
			if = {
				limit = { ROOT = { has_country_flag = iop_roman_empire } }
				custom_effect_tooltip = iop_establish_north_africa_roman_tt
			}
			else = {
				custom_effect_tooltip = iop_establish_north_africa_tt
			}
			custom_effect_tooltip = iop_directorate_zone_tt
			hidden_effect = {
				set_country_flag = iop_north_africa_established
				448 = { add_core_of = INA }
				661 = { add_core_of = INA }
				449 = { add_core_of = INA }
				662 = { add_core_of = INA }
				450 = { add_core_of = INA }
				451 = { add_core_of = INA }
				663 = { add_core_of = INA }
				273 = { add_core_of = INA }
				release_puppet = INA
				if = {
					limit = { ROOT = { has_country_flag = iop_roman_empire } }
					set_autonomy = {
						target = INA
						autonomous_state = autonomy_province
					}
					INA = { set_cosmetic_tag = INA_roman }
				}
				else = {
					set_autonomy = {
						target = INA
						autonomous_state = autonomy_military_occupation
					}
				}
				if = {
					limit = { 448 = { iop_med_controlled = yes } }
					INA = { transfer_state = 448 }
				}
				if = {
					limit = { 661 = { iop_med_controlled = yes } }
					INA = { transfer_state = 661 }
				}
				if = {
					limit = { 449 = { iop_med_controlled = yes } }
					INA = { transfer_state = 449 }
				}
				if = {
					limit = { 662 = { iop_med_controlled = yes } }
					INA = { transfer_state = 662 }
				}
				if = {
					limit = { 450 = { iop_med_controlled = yes } }
					INA = { transfer_state = 450 }
				}
				if = {
					limit = { 451 = { iop_med_controlled = yes } }
					INA = { transfer_state = 451 }
				}
				if = {
					limit = { 663 = { iop_med_controlled = yes } }
					INA = { transfer_state = 663 }
				}
				if = {
					limit = { 273 = { iop_med_controlled = yes } }
					INA = { transfer_state = 273 }
				}
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	# ------------------------------------------------------------

	# Each decision needs INA to exist and Italy to control at
	# least one state of the group. No border checks, no events.
	# ------------------------------------------------------------

			}
		}
		available = {

			OR = {
				controls_state = 665
				controls_state = 458
			}
		}
		days_re_enable = 3
		complete_effect = {
			if = {
				limit = { controls_state = 665 }
				665 = { add_core_of = INA }
				INA = { transfer_state = 665 }
			}
			if = {
				limit = { controls_state = 458 }
				458 = { add_core_of = INA }
				INA = { transfer_state = 458 }
			}
		}
		ai_will_do = { factor = 0 }
	}

	# SPECIAL: SPANISH AFRICA (290)
	# v0.0.32: fires an event - give it to North Africa (INA) or

	# ------------------------------------------------------------

	iop_decide_290_spanish_africa = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 290

		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA

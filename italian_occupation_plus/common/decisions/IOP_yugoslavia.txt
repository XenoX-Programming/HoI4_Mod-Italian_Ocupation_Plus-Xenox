# ============================================================
# Italian Occupation Plus v0.1 - Yugoslavia
# Works like Reichskommissariats Plus, but for Italy:
# 1) As Italy, occupy Yugoslav land.
# 2) Found ICR / ISE / IMT with the "Establish" decisions.
#    Founding requires controlling the capital state only.
#    Any other initial states you control transfer automatically.
# 3) Use "Determine fate of ..." decisions to assign further
#    states via event (choose Croatia / Serbia / Montenegro).
# 4) Zara (163) and Istria (852) are special: give to Croatia
#    OR annex to Italy (adds Italian core).
# ============================================================

# NOTE v0.1.1: vanilla "political_actions" category, guaranteed to show up.

		icon = generic_political_pressure

			highlight_state_targets = {
				state = 103
				state = 109
				state = 104
				state = 804
			}
		}

		allowed = {
			original_tag = ITA
		}

		visible = {
			original_tag = ITA
			NOT = { country_exists = ICR }
		}

		available = {
			controls_state = 109
		}

		fire_only_once = no

		complete_effect = {
			custom_effect_tooltip = iop_establish_croatia_tt
			hidden_effect = {
				set_country_flag = iop_croatia_established
				103 = { add_core_of = ICR }
				109 = { add_core_of = ICR }
				104 = { add_core_of = ICR }
				804 = { add_core_of = ICR }
				release_puppet = ICR
				set_autonomy = {
					target = ICR
					autonomous_state = autonomy_military_occupation
				}
				if = {
					limit = { controls_state = 103 }
					ICR = { transfer_state = 103 }
				}
				if = {
					limit = { controls_state = 109 }
					ICR = { transfer_state = 109 }
				}
				if = {
					limit = { controls_state = 104 }
					ICR = { transfer_state = 104 }
				}
				if = {
					limit = { controls_state = 804 }
					ICR = { transfer_state = 804 }
				}
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

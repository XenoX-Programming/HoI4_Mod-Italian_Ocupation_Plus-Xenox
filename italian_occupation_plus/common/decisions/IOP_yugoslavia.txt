# ============================================================
# Italian Occupation Plus v0.1 - Yugoslavia
# Works like Reichskommissariats Plus, but for Italy:
# 1) As Italy, occupy Yugoslav land.
# 2) Found ICR / ISE / IMT with the "Establish" decisions.
#    Founding requires the capital state to be controlled by Italy or one of its subjects.
#    Any other initial states controlled by Italy or its subjects transfer automatically.
# 3) Use "Determine fate of ..." decisions to assign further
#    states via event (choose Croatia / Serbia / Montenegro).
# 4) Zara (163) and Istria (852) are special: give to Croatia
#    OR annex to Italy (adds Italian core).
# ============================================================

		icon = generic_political_pressure

		highlight_states = {
			highlight_state_targets = {
				state = 109

		}

		allowed = {
			original_tag = ITA
		}

		visible = {
			original_tag = ITA
			NOT = { country_exists = ICR }
			# Only appears once Italy or its subjects control at least half of the required territory (2 of 3 states).
			count_triggers = {
				amount = 2
				109 = { iop_med_controlled = yes }
				104 = { iop_med_controlled = yes }
				804 = { iop_med_controlled = yes }

		}

		fire_only_once = no

		complete_effect = {
			hidden_effect = {
				iop_grant_zone_croatia = yes

			if = {
				limit = { ROOT = { has_country_flag = iop_roman_empire } }
				custom_effect_tooltip = iop_establish_croatia_roman_tt
			}
			else = {
				custom_effect_tooltip = iop_establish_croatia_tt
			}
			custom_effect_tooltip = iop_directorate_zone_tt
			hidden_effect = {
				set_country_flag = iop_croatia_established
				109 = { add_core_of = ICR }
				104 = { add_core_of = ICR }
				804 = { add_core_of = ICR }
				release_puppet = ICR
				if = {
					limit = { ROOT = { has_country_flag = iop_roman_empire } }
					set_autonomy = {
						target = ICR
						autonomous_state = autonomy_province
					}
					ICR = { set_cosmetic_tag = ICR_roman }
				}
				else = {
					set_autonomy = {
						target = ICR
						autonomous_state = autonomy_military_occupation
					}
				}
				if = {
					limit = { 109 = { iop_med_controlled = yes } }
					ICR = { transfer_state = 109 }
				}
				if = {
					limit = { 104 = { iop_med_controlled = yes } }
					ICR = { transfer_state = 104 }
				}
				if = {
					limit = { 804 = { iop_med_controlled = yes } }
					ICR = { transfer_state = 804 }
				}
			}
		}

		ai_will_do = {
			factor = 0
		}

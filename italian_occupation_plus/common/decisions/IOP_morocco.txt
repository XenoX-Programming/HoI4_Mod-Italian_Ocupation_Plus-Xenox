#    - Spanish Africa (290): North Africa (INA), Morocco (IMR) or Italy
#    - Sidi Ifni (783):      North Africa (INA), Morocco (IMR) or Italy

iop_occupation = {

	# ------------------------------------------------------------
	# FOUNDING DECISION
	# ------------------------------------------------------------

	iop_establish_morocco = {
		icon = generic_political_pressure

		highlight_states = {
			highlight_state_targets = {
				state = 461
				state = 462
			}
		}

		allowed = {
			original_tag = ITA
		}

		visible = {
			original_tag = ITA
			NOT = { country_exists = IMR }
			# Only appears once Italy or its subjects control at least half of the
			# required territory (1 of 2 states).
			count_triggers = {
				amount = 1
				461 = { iop_med_controlled = yes }
				462 = { iop_med_controlled = yes }
			}
		}

		available = {
			461 = { iop_med_controlled = yes }
		}

		fire_only_once = no

		complete_effect = {
			hidden_effect = {
				iop_grant_zone_morocco = yes
			}
			if = {
				limit = { ROOT = { has_country_flag = iop_roman_empire } }
				custom_effect_tooltip = iop_establish_morocco_roman_tt
			}
			else = {
				custom_effect_tooltip = iop_establish_morocco_tt
			}
			custom_effect_tooltip = iop_directorate_zone_tt
			hidden_effect = {
				set_country_flag = iop_morocco_established
				461 = { add_core_of = IMR }
				462 = { add_core_of = IMR }
				release_puppet = IMR
				if = {
					limit = { ROOT = { has_country_flag = iop_roman_empire } }
					set_autonomy = {
						target = IMR
						autonomous_state = autonomy_province
					}
					IMR = { set_cosmetic_tag = IMR_roman }
				}
				else = {
					set_autonomy = {
						target = IMR
						autonomous_state = autonomy_military_occupation
					}
				}
				if = {
					limit = { 461 = { iop_med_controlled = yes } }
					IMR = { transfer_state = 461 }
				}
				if = {
					limit = { 462 = { iop_med_controlled = yes } }
					IMR = { transfer_state = 462 }
				}
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	# ------------------------------------------------------------
	# EXPANSION: RIO DE ORO (699)
	# North Africa (INA), Morocco (IMR) - existence check only, the
	# coast is considered contiguous - or annex to Italy.
	# ------------------------------------------------------------

	iop_decide_699_rio_de_oro = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 699
			}
		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA
			controls_state = 699
			# Hide once integrated into Italy (the annex option adds an Italian core).
			NOT = { 699 = { is_core_of = ITA } }
			OR = {
				country_exists = INA
				country_exists = IMR
			}
		}
		available = {
			controls_state = 699
			OR = {
				country_exists = INA
				country_exists = IMR
			}
		}
		days_re_enable = 3
		complete_effect = {
			country_event = { id = iop_africa.699 }
		}
		ai_will_do = { factor = 0 }
	}
}

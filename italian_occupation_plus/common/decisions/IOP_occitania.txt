# ============================================================
# Italian Occupation Plus v0.0.20 - Occitania
# 1) Found IOC with the "Establish" decision (needs 19 Aquitaine / Bordeaux).
# 2) "Fate of Provence" (32 Alpes + 21 Bouches-du-Rhone): IOC or Italy.
# 3) "Fate of Savoy and the Riviera" (735 Savoy + 851 Var): IOC or Italy.
#    Both need IOC to OWN a state bordering one of the group states.
# 4) "Fate of Corsica" (1): island - IOC only needs to exist.
# 5) "Transfer Poitou to Occitania" (23): direct, no event.
# ============================================================

iop_occupation = {

	# ------------------------------------------------------------
	# FOUNDING DECISION
	# ------------------------------------------------------------

	iop_establish_occitania = {
		icon = generic_political_pressure

		highlight_states = {
			highlight_state_targets = {
				state = 33
				state = 25
				state = 19
				state = 806
				state = 31
				state = 26
				state = 20
				state = 22
			}
		}

		allowed = {
			original_tag = ITA
		}

		visible = {
			original_tag = ITA
			NOT = { country_exists = IOC }
			# Only appears once Italy or its subjects control at least half of the required territory (4 of 8 states).
			count_triggers = {
				amount = 4
				33 = { iop_med_controlled = yes }
				25 = { iop_med_controlled = yes }
				19 = { iop_med_controlled = yes }
				806 = { iop_med_controlled = yes }
				31 = { iop_med_controlled = yes }
				26 = { iop_med_controlled = yes }
				20 = { iop_med_controlled = yes }
				22 = { iop_med_controlled = yes }
			}
		}

		available = {
			19 = { iop_med_controlled = yes }
		}

		fire_only_once = no

		complete_effect = {
			hidden_effect = {
				iop_grant_zone_occitania = yes
			}
			custom_effect_tooltip = iop_establish_occitania_tt
			custom_effect_tooltip = iop_directorate_zone_tt
			hidden_effect = {
				country_event = { id = iop_gov.12 days = 2 }
			}
			hidden_effect = {
				set_country_flag = iop_occitania_established
				33 = { add_core_of = IOC }
				25 = { add_core_of = IOC }
				19 = { add_core_of = IOC }
				806 = { add_core_of = IOC }
				31 = { add_core_of = IOC }
				26 = { add_core_of = IOC }
				20 = { add_core_of = IOC }
				22 = { add_core_of = IOC }
				release_puppet = IOC
				set_autonomy = {
					target = IOC
					autonomous_state = autonomy_military_occupation
				}
				if = {
					limit = { 33 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 33 }
				}
				if = {
					limit = { 25 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 25 }
				}
				if = {
					limit = { 19 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 19 }
				}
				if = {
					limit = { 806 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 806 }
				}
				if = {
					limit = { 31 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 31 }
				}
				if = {
					limit = { 26 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 26 }
				}
				if = {
					limit = { 20 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 20 }
				}
				if = {
					limit = { 22 = { iop_med_controlled = yes } }
					IOC = { transfer_state = 22 }
				}
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	# ------------------------------------------------------------
	# SPECIAL: PROVENCE - Alpes (32) + Bouches-du-Rhone (21)
	# IOC (needs to own a bordering state) or Italy.
	# ------------------------------------------------------------

	iop_decide_32_provence = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 32
				state = 21
			}
		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA
			# At least one state must still be undecided (controlled but not an Italian core yet).
			OR = {
				AND = {
					controls_state = 32
					NOT = { 32 = { is_core_of = ITA } }
				}
				AND = {
					controls_state = 21
					NOT = { 21 = { is_core_of = ITA } }
				}
			}
			country_exists = IOC
		}
		available = {
			OR = {
				controls_state = 32
				controls_state = 21
			}
			OR = {
				32 = { any_neighbor_state = { is_owned_by = IOC } }
				21 = { any_neighbor_state = { is_owned_by = IOC } }
			}
		}
		days_re_enable = 3
		complete_effect = {
			country_event = { id = iop_occitania.32 }
		}
		ai_will_do = { factor = 0 }
	}

	# ------------------------------------------------------------
	# SPECIAL: SAVOY (735) + VAR (851)
	# IOC (needs to own a bordering state) or Italy.
	# ------------------------------------------------------------

	iop_decide_735_savoy_var = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 735
				state = 851
			}
		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA
			# At least one state must still be undecided (controlled but not an Italian core yet).
			OR = {
				AND = {
					controls_state = 735
					NOT = { 735 = { is_core_of = ITA } }
				}
				AND = {
					controls_state = 851
					NOT = { 851 = { is_core_of = ITA } }
				}
			}
			country_exists = IOC
		}
		available = {
			OR = {
				controls_state = 735
				controls_state = 851
			}
			OR = {
				735 = { any_neighbor_state = { is_owned_by = IOC } }
				851 = { any_neighbor_state = { is_owned_by = IOC } }
			}
		}
		days_re_enable = 3
		complete_effect = {
			country_event = { id = iop_occitania.735 }
		}
		ai_will_do = { factor = 0 }
	}

	# ------------------------------------------------------------
	# SPECIAL: CORSICA (1) - island: IOC only needs to exist.
	# ------------------------------------------------------------

	iop_decide_1_corsica = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 1
			}
		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA
			controls_state = 1
			# Hide once integrated into Italy (the annex option adds an Italian core).
			NOT = { 1 = { is_core_of = ITA } }
			country_exists = IOC
		}
		available = {
			controls_state = 1
			country_exists = IOC
		}
		days_re_enable = 3
		complete_effect = {
			country_event = { id = iop_occitania.1 }
		}
		ai_will_do = { factor = 0 }
	}

	# ------------------------------------------------------------
	# DIRECT TRANSFER - Poitou (23) -> IOC. Algeria pattern: no event,
	# needs IOC to exist and Italy or one of its subjects to own/control
	# the state.
	# ------------------------------------------------------------

	# ------------------------------------------------------------
	# v0.0.33 - AQUITAINE (19) + PYRENEES-ATLANTIQUES (806)
	# Direct transfer to the Occitan occupation government (every
	# controlled state of the group transfers). Self-hiding: once
	# neither state is under Italian control the decision disappears.
	# ------------------------------------------------------------

	iop_give_aquitaine = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 19
				state = 806
			}
		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA
			country_exists = IOC
			OR = {
				controls_state = 19
				controls_state = 806
			}
		}
		available = {
			country_exists = IOC
			OR = {
				controls_state = 19
				controls_state = 806
			}
		}
		days_re_enable = 3
		complete_effect = {
			if = {
				limit = { controls_state = 19 }
				19 = { add_core_of = IOC }
				IOC = { transfer_state = 19 }
			}
			if = {
				limit = { controls_state = 806 }
				806 = { add_core_of = IOC }
				IOC = { transfer_state = 806 }
			}
		}
		ai_will_do = { factor = 0 }
	}

	iop_give_poitou = {
		icon = generic_decision

		highlight_states = {
			highlight_state_targets = {
				state = 23
			}
		}
		allowed = { original_tag = ITA }
		visible = {
			original_tag = ITA
			country_exists = IOC
			# One-shot: the transfer itself keeps "owner is a subject"
			# true, so without this flag the decision would reappear.
			NOT = { has_country_flag = iop_poitou_given }
			OR = {
				controls_state = 23
				owns_state = 23
				23 = { owner = { is_subject_of = ROOT } }
			}
		}
		available = {
			country_exists = IOC
			OR = {
				controls_state = 23
				owns_state = 23
				23 = { owner = { is_subject_of = ROOT } }
			}
		}
		fire_only_once = yes
		days_re_enable = 3
		complete_effect = {
			hidden_effect = {
				set_country_flag = iop_poitou_given
			}
			23 = { add_core_of = IOC }
			IOC = { transfer_state = 23 }
		}
		ai_will_do = { factor = 0 }
	}

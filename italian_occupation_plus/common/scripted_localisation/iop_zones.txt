# Italian Occupation Plus (v0.0.35)
# Scripted localisation: the Occupation Directorate national spirit "contains"
# the per-zone dynamic modifiers by listing exactly the zones Italy has
# founded. Each defined_text below resolves to that zone's list line
# (iop_zone_list_*, defined in IOP_zones_l_english.yml, trailing newline
# included) when the zone flag exists, or to the empty string otherwise, so
# the hub description only ever shows active zones. Scope at render time is
# the idea holder (Italy), hence has_country_flag.

defined_text = {
	name = Iop_Zone_List_Croatia
	text = {
		trigger = { has_country_flag = iop_zone_croatia }
		localization_key = iop_zone_list_croatia
	}
	text = {
		localization_key = iop_zone_empty
	}
}

# A Mediterranean coastal state held by Italy itself or by one of
# Italy's subjects (occupation governments). Used by the
# "Restore the Roman Empire" decision.
iop_med_owned = {
	OR = {
		is_owned_by = ITA
		OWNER = { is_subject_of = ITA }
	}
}

# A state controlled by Italy itself or by one of Italy's subjects
# (occupation governments). Used by the founding decisions, so that
# territory under puppet occupation counts toward founding a new
# occupation government.
iop_med_controlled = {
	OR = {
		is_controlled_by = ITA
		CONTROLLER = { is_subject_of = ITA }
	}
}

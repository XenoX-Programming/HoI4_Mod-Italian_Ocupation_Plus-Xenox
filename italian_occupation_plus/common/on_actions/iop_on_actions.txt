# ============================================================
# Italian Occupation Plus - on_actions
# Handles dynamic status conversion for Italian subjects (such as
# vanilla Italian East Africa AOI) if formed or puppeted after the
# Roman Empire has already been restored.
# ============================================================

on_actions = {
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					original_tag = ITA
					has_country_flag = iop_roman_empire
				}
				if = {
					limit = {
						country_exists = AOI
						AOI = {
							is_subject_of = ROOT
							NOT = { has_cosmetic_tag = AOI_roman }
						}
					}
					set_autonomy = {
						target = AOI
						autonomous_state = autonomy_province
					}
					AOI = { set_cosmetic_tag = AOI_roman }
				}
			}
		}
	}
}

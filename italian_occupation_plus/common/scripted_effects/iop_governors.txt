# ============================================================
# Italian Occupation Plus - governor appointment effects (v0.0.37)
#
# Called by the iop_gov.<n> governor appointment events (ROOT = Italy).
# Every occupation government has three effects, one per style:
#
#   iop_appoint_<zone>_order   Italian general / proconsul (incumbent)
#   iop_appoint_<zone>_legit   local collaborator, monarch or cleric
#   iop_appoint_<zone>_party   Fascist Party appointee
#
# An appointment clears the zone's previous style (handing its Italy-side
# contribution back), installs the new governor in the puppet for all four
# ideologies, swaps the puppet's governor spirit, and updates Rome's
# variable-scaled governor dynamic modifiers. All of it is hidden: the
# event option carries the tooltip.
# ============================================================

# ------------------------------------------------ Rome-side plumbing
# One dynamic modifier per style, scaled by these variables, so the
# Directorate's character grows with every zone run a given way.

iop_gov_add_order = {
	if = { limit = { NOT = { has_variable = iop_gov_order_garrison } } set_variable = { iop_gov_order_garrison = 0 } }
	if = { limit = { NOT = { has_variable = iop_gov_order_resistance } } set_variable = { iop_gov_order_resistance = 0 } }
	subtract_from_variable = { iop_gov_order_garrison = 0.01 }
	subtract_from_variable = { iop_gov_order_resistance = 0.01 }
	add_dynamic_modifier = { modifier = iop_gov_order }
	force_update_dynamic_modifier = yes
}

iop_gov_drop_order = {
	if = { limit = { has_variable = iop_gov_order_garrison } add_to_variable = { iop_gov_order_garrison = 0.01 } }
	if = { limit = { has_variable = iop_gov_order_resistance } add_to_variable = { iop_gov_order_resistance = 0.01 } }
	force_update_dynamic_modifier = yes
}

iop_gov_add_legit = {
	if = { limit = { NOT = { has_variable = iop_gov_legit_compliance } } set_variable = { iop_gov_legit_compliance = 0 } }
	if = { limit = { NOT = { has_variable = iop_gov_legit_autonomy } } set_variable = { iop_gov_legit_autonomy = 0 } }
	add_to_variable = { iop_gov_legit_compliance = 0.01 }
	add_to_variable = { iop_gov_legit_autonomy = 0.03 }
	add_dynamic_modifier = { modifier = iop_gov_legit }
	force_update_dynamic_modifier = yes
}

iop_gov_drop_legit = {
	if = { limit = { has_variable = iop_gov_legit_compliance } subtract_from_variable = { iop_gov_legit_compliance = 0.01 } }
	if = { limit = { has_variable = iop_gov_legit_autonomy } subtract_from_variable = { iop_gov_legit_autonomy = 0.03 } }
	force_update_dynamic_modifier = yes
}

iop_gov_add_party = {
	if = { limit = { NOT = { has_variable = iop_gov_party_pp } } set_variable = { iop_gov_party_pp = 0 } }
	if = { limit = { NOT = { has_variable = iop_gov_party_resources } } set_variable = { iop_gov_party_resources = 0 } }
	if = { limit = { NOT = { has_variable = iop_gov_party_resistance } } set_variable = { iop_gov_party_resistance = 0 } }
	add_to_variable = { iop_gov_party_pp = 0.02 }
	add_to_variable = { iop_gov_party_resources = 0.01 }
	add_to_variable = { iop_gov_party_resistance = 0.01 }
	add_dynamic_modifier = { modifier = iop_gov_party }
	force_update_dynamic_modifier = yes
}

iop_gov_drop_party = {

	if = { limit = { has_variable = iop_gov_party_resistance } subtract_from_variable = { iop_gov_party_resistance = 0.01 } }
	force_update_dynamic_modifier = yes
}

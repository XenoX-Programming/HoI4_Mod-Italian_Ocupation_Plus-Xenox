# ============================================================
# Italian Occupation Plus - zone-grant scripted effects (v0.0.34)
# Called once by every founding (17 founding decisions + the
# Montenegro and Oman event options). Each effect ensures Italy
# holds the Occupation Directorate hub spirit, then grants that
# zone's own dynamic modifier - once ever per zone (flag-guarded,
# so re-founding a destroyed puppet grants nothing new).
#
# Usage (inside hidden_effect):  iop_grant_zone_croatia = yes
# ============================================================

iop_grant_zone_croatia = {
	if = {
		limit = { NOT = { has_idea = iop_occupation_directorate } }
		add_ideas = iop_occupation_directorate
	}
	if = {
		limit = { NOT = { has_country_flag = iop_zone_croatia } }
		set_country_flag = iop_zone_croatia
		add_dynamic_modifier = { modifier = iop_zone_croatia }
	}
}

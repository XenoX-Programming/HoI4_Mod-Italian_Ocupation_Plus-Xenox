# ---------------- 731 - Central Macedonia ----------------

	id = iop_greece.731
	title = iop_yugo_fate_title
	desc = iop_greece_731_desc
	picture = GFX_report_event_generic
	is_triggered_only = yes
	option = {
		name = iop_give_to_IGR

			country_exists = IGR

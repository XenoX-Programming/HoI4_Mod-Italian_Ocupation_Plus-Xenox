add_namespace = iop_africa

# ---------------- 783 - Sidi Ifni (SPECIAL) ----------------
# Give to INA or IMR, or integrate into Italy proper (adds Italian core).
country_event = {
	id = iop_africa.783
	title = iop_africa_ifni_title
	desc = iop_africa_783_desc
	picture = GFX_report_event_generic
	is_triggered_only = yes
	option = {
		name = iop_give_ifni_to_INA
		trigger = {
			country_exists = INA
		}
		783 = { add_core_of = INA }
		INA = { transfer_state = 783 }
		ai_chance = { factor = 0 }
	}
	option = {
		name = iop_give_ifni_to_IMR
		trigger = {
			country_exists = IMR
		}
		783 = { add_core_of = IMR }
		IMR = { transfer_state = 783 }
		ai_chance = { factor = 0 }
	}
	option = {
		name = iop_annex_to_italy
		custom_effect_tooltip = iop_annex_to_italy_tt
		hidden_effect = {
			783 = { add_core_of = ITA }
			ITA = { transfer_state = 783 }
		}
		ai_chance = { factor = 100 }
	}
}

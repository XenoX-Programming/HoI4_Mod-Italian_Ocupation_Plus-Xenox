# ============================================================

add_namespace = iop_spain

# ---------------- 118 - Fall of Gibraltar (SPECIAL) ----------------
# Give Gibraltar to ISP (needs ISP to own a neighbor) OR integrate into Italy.
country_event = {
	id = iop_spain.118
	title = iop_spain_gibraltar_title
	desc = iop_spain_118_desc
	picture = GFX_report_event_generic
	is_triggered_only = yes
	option = {
		name = iop_give_gibraltar_to_ISP
		trigger = {
			country_exists = ISP
			118 = { any_neighbor_state = { is_owned_by = ISP } }
		}
		118 = { add_core_of = ISP }
		ISP = { transfer_state = 118 }
		ai_chance = { factor = 0 }
	}
	option = {
		name = iop_annex_to_italy
		custom_effect_tooltip = iop_annex_to_italy_tt
		hidden_effect = {
			118 = { add_core_of = ITA }
			ITA = { transfer_state = 118 }
		}
		ai_chance = { factor = 100 }
	}
}

# ============================================================
# Italian Occupation Plus v0.0.29 - flavour events.
# ROOT = Italy. Each event references a named puppet that must
# already exist, so it only ever fires mid-occupation. Effects
# hand out the timed iop_spirit_* ideas or small instant bonuses.
# ============================================================

add_namespace = iop_flavour

# ---------------- 1 - Balkan partisan ambush (Croatia) ----------------
country_event = {
	id = iop_flavour.1
	title = iop_flavour_1_title
	desc = iop_flavour_1_desc
	picture = GFX_report_event_generic
	is_triggered_only = no
	mean_time_to_happen = { months = 30 }
	trigger = {
		original_tag = ITA
		country_exists = ICR
		NOT = { has_country_flag = iop_flavour_cooldown }
	}
	immediate = { set_country_flag = { flag = iop_flavour_cooldown days = 300 value = 1 } }
	option = {
		name = iop_flavour_1_a
		ICR = { add_timed_idea = { idea = iop_spirit_reprisals days = 180 } }
		ai_chance = { factor = 50 }
	}
	option = {
		name = iop_flavour_1_b
		ICR = { add_timed_idea = { idea = iop_spirit_conciliation days = 180 } }
		ai_chance = { factor = 50 }
	}
}

	}

		name = iop_flavour_7_b

		ai_chance = { factor = 40 }
	}
}

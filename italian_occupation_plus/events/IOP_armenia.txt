# ---------------- 354 + 800 - Trabzon + Van: ITR / IAM ----------------

	id = iop_armenia.354
	title = iop_armenia_trabzon_van_title
	desc = iop_armenia_354_desc

	option = {
		name = iop_give_trabzon_van_to_ITR

			limit = { controls_state = 354 }

			limit = { controls_state = 800 }
			800 = { add_core_of = ITR }
			ITR = { transfer_state = 800 }
		}

	}
	option = {
		name = iop_give_trabzon_van_to_IAM
		trigger = {
			country_exists = IAM
			OR = {
				354 = { any_neighbor_state = { is_owned_by = IAM } }
				800 = { any_neighbor_state = { is_owned_by = IAM } }
			}
		}
		if = {
			limit = { controls_state = 354 }

			IAM = { transfer_state = 354 }

		if = {
			limit = { controls_state = 800 }
			800 = { add_core_of = IAM }
			IAM = { transfer_state = 800 }
		}
		ai_chance = { factor = 0 }
	}

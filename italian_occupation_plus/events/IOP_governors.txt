# ============================================================
# Italian Occupation Plus - governor appointment events (v0.0.37)
#
# One event per occupation government. Fired when the government is
# founded (from the iop_establish_<zone> decisions, or the Montenegrin
# and Omani founding events) and again whenever Italy replaces a
# governor from the Governorates decision tab. ROOT = Italy.
#
# Three governing styles, each with its own man and its own price:
#   a  Italian general  - order: cheaper garrisons, broken resistance
#   b  local figure     - legitimacy: compliance now, autonomy drift later
#   c  Party appointee  - exploitation: political power, resources, resentment
# ============================================================

add_namespace = iop_gov

# ---------------- 1 - Croatia (ICR) ----------------
country_event = {
	id = iop_gov.1
	title = iop_gov_1_title
	desc = iop_gov_1_desc
	picture = GFX_report_event_generic
	is_triggered_only = yes

	option = { # order
		name = iop_gov_1_a
		trigger = { country_exists = ICR }
		hidden_effect = { iop_appoint_croatia_order = yes }
		custom_effect_tooltip = iop_gov_order_tt
		ai_chance = { factor = 45 }
	}

	option = { # legit
		name = iop_gov_1_b
		trigger = { country_exists = ICR }
		hidden_effect = { iop_appoint_croatia_legit = yes }
		custom_effect_tooltip = iop_gov_legit_tt
		ai_chance = { factor = 30 }
	}

	option = { # party
		name = iop_gov_1_c
		trigger = { country_exists = ICR }
		hidden_effect = { iop_appoint_croatia_party = yes }
		custom_effect_tooltip = iop_gov_party_tt
		ai_chance = { factor = 25 }
	}
}
